    add_compile_options(/utf-8)

add_library(daymark_update_sdk STATIC sdk/Update.cpp)
target_include_directories(daymark_update_sdk PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/.." "${PROJECT_SOURCE_DIR}/third_party/update/json")
target_compile_features(daymark_update_sdk PUBLIC cxx_std_20)
if(WIN32)
    enable_language(RC)
    configure_file(frontends/win32/manifest.rc.in "${CMAKE_CURRENT_BINARY_DIR}/update-manifest.rc" @ONLY)
    set(DAYMARK_UPDATE_PUBLIC_KEYS "${PROJECT_SOURCE_DIR}/config/update-public-keys.json" CACHE FILEPATH "Public update trust keys JSON; never a private signing key")
    set(update_key_count 0)
    set(update_key_entries "")
    if(DAYMARK_UPDATE_PUBLIC_KEYS)
        file(READ "${DAYMARK_UPDATE_PUBLIC_KEYS}" update_key_json)
        string(JSON update_key_count LENGTH "${update_key_json}" keys)
        if(update_key_count GREATER 0)
            math(EXPR update_key_last "${update_key_count}-1")
            foreach(i RANGE 0 ${update_key_last})
                string(JSON update_key_id GET "${update_key_json}" keys ${i} key_id)
                string(JSON update_key_xy GET "${update_key_json}" keys ${i} public_xy)
                string(LENGTH "${update_key_xy}" update_key_length)
                if(NOT update_key_id MATCHES "^[a-zA-Z0-9_-]+$" OR NOT update_key_xy MATCHES "^[0-9a-f]+$" OR NOT update_key_length EQUAL 128)
                    message(FATAL_ERROR "Invalid public update trust key")
                endif()
                string(APPEND update_key_entries "{\"${update_key_id}\",\"${update_key_xy}\"},\n")
            endforeach()
        endif()
    endif()
    file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/UpdateTrust.h" "#pragma once\n#include <array>\n#include <string_view>\n#include <utility>\ninline constexpr std::array<std::pair<std::string_view,std::string_view>,${update_key_count}> BuildTrustKeys{{${update_key_entries}}};\n")
    file(READ "${PROJECT_SOURCE_DIR}/docs/api/diagnostics.json" setup_codes_json)
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${PROJECT_SOURCE_DIR}/docs/api/diagnostics.json")
    string(JSON client_codes_count LENGTH "${setup_codes_json}" client)
    math(EXPR client_codes_last "${client_codes_count}-1")
    set(setup_codes_entries "")
    set(setup_codes_count 0)
    foreach(i RANGE 0 ${client_codes_last})
        string(JSON key MEMBER "${setup_codes_json}" client ${i})
        if(key MATCHES "^setup[.]")
            string(JSON code GET "${setup_codes_json}" client "${key}")
            string(APPEND setup_codes_entries "{\"${key}\",\"${code}\"},\n")
            math(EXPR setup_codes_count "${setup_codes_count}+1")
        endif()
    endforeach()
    file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/SetupDiagnosticCodes.h" "#pragma once\n#include <array>\n#include <string_view>\n#include <utility>\ninline constexpr std::array<std::pair<std::string_view,std::string_view>,${setup_codes_count}> BuildSetupCodes{{${setup_codes_entries}}};\n")
    set(update_miniz "${PROJECT_SOURCE_DIR}/third_party/update/miniz")
    set(update_miniz_sources "${update_miniz}/miniz.c" "${update_miniz}/miniz_tinfl.c" "${update_miniz}/miniz_zip.c")
    set_source_files_properties(${update_miniz_sources} PROPERTIES LANGUAGE CXX)
    add_library(daymark_update_zip STATIC ${update_miniz_sources})
    target_include_directories(daymark_update_zip PUBLIC "${update_miniz}")
    target_compile_definitions(daymark_update_zip PUBLIC MINIZ_NO_ZLIB_APIS MINIZ_NO_DEFLATE_APIS)
    add_library(daymark_update_windows STATIC platform/windows/Windows.cpp platform/windows/Transport.cpp platform/windows/Engine.cpp platform/windows/Setup.cpp)
    target_include_directories(daymark_update_windows PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
    target_link_libraries(daymark_update_windows PUBLIC daymark_update_sdk PRIVATE daymark_update_zip bcrypt winhttp advapi32 shell32)
    target_compile_definitions(daymark_update_windows PUBLIC UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
    add_executable(daymark-update-cli cli/Main.cpp)
    target_link_libraries(daymark-update-cli PRIVATE daymark_update_windows)
    add_executable(daymark-bootstrap WIN32 frontends/win32/Bootstrap.cpp)
    add_executable(daymark-updater WIN32 frontends/win32/Updater.cpp)
    target_link_libraries(daymark-bootstrap PRIVATE daymark_update_windows)
    target_link_libraries(daymark-updater PRIVATE daymark_update_windows comctl32)
    add_executable(daymark-setup-helper frontends/win32/SetupMain.cpp)
    target_link_libraries(daymark-setup-helper PRIVATE daymark_update_windows)
    foreach(update_host daymark-update-cli daymark-bootstrap daymark-updater daymark-setup-helper)
        target_sources(${update_host} PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/update-manifest.rc")
        if(MSVC)
            # The reviewed asInvoker/DPI manifest is already embedded as resource 1.
            # MSVC's generated manifest would otherwise duplicate the same resource.
            target_link_options(${update_host} PRIVATE /MANIFEST:NO)
        endif()
    endforeach()
    set_target_properties(daymark-bootstrap PROPERTIES OUTPUT_NAME daymark-launcher)
    if(BUILD_TESTING)
        add_executable(test_update_windows "${PROJECT_SOURCE_DIR}/tests/update/UpdateWindowsTest.cpp")
        target_link_libraries(test_update_windows PRIVATE daymark_update_windows)
        if(MINGW)
            target_link_options(test_update_windows PRIVATE -static)
        endif()
        add_test(NAME update_windows_crypto COMMAND test_update_windows)
        set_tests_properties(update_windows_crypto PROPERTIES LABELS "fast;update;windows")
    endif()
    if(MINGW)
        foreach(update_host daymark-update-cli daymark-bootstrap daymark-updater daymark-setup-helper)
            target_link_options(${update_host} PRIVATE -static -municode)
        endforeach()
    elseif(MSVC)
        set_property(TARGET daymark_update_sdk daymark_update_windows daymark_update_zip daymark-update-cli daymark-bootstrap daymark-updater PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    endif()
    if(BUILD_TESTING)
        add_executable(test_setup_helper frontends/win32/SetupMain.cpp)
        target_compile_definitions(test_setup_helper PRIVATE DAYMARK_UPDATE_FIXTURE)
        target_link_libraries(test_setup_helper PRIVATE daymark_update_windows)
        if(MINGW)
            target_link_options(test_setup_helper PRIVATE -static -municode)
        endif()
        add_executable(test_update_cli cli/Main.cpp)
        target_compile_definitions(test_update_cli PRIVATE DAYMARK_UPDATE_FIXTURE)
        target_link_libraries(test_update_cli PRIVATE daymark_update_windows)
        if(MINGW)
            target_link_options(test_update_cli PRIVATE -static -municode)
        endif()
        add_executable(test_update_bootstrap WIN32 frontends/win32/Bootstrap.cpp)
        target_compile_definitions(test_update_bootstrap PRIVATE DAYMARK_UPDATE_FIXTURE)
        target_link_libraries(test_update_bootstrap PRIVATE daymark_update_windows)
        if(MINGW)
            target_link_options(test_update_bootstrap PRIVATE -static -municode)
        endif()
        foreach(fixture_version old new)
            add_executable(test_update_payload_${fixture_version} "${PROJECT_SOURCE_DIR}/tests/update/FixturePayload.cpp")
            target_link_libraries(test_update_payload_${fixture_version} PRIVATE daymark_update_windows)
            if(fixture_version STREQUAL "old")
                target_compile_definitions(test_update_payload_${fixture_version} PRIVATE FIXTURE_VERSION="2.0.0")
            else()
                target_compile_definitions(test_update_payload_${fixture_version} PRIVATE FIXTURE_VERSION="2.0.1")
            endif()
            if(MINGW)
                target_link_options(test_update_payload_${fixture_version} PRIVATE -static -municode)
            endif()
        endforeach()
    endif()
    if(MSVC)
        foreach(update_target daymark_update_sdk daymark_update_windows daymark_update_zip daymark-update-cli daymark-bootstrap daymark-updater daymark-setup-helper test_setup_helper test_update_windows test_update_cli test_update_bootstrap test_update_payload_old test_update_payload_new)
            if(TARGET ${update_target})
                set_property(TARGET ${update_target} PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
            endif()
        endforeach()
    endif()
endif()
